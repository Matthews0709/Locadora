CREATE DATABASE locadora
GO
USE locadora
GO

CREATE TABLE categoria (
    id BIGINT AUTO_INCREMENT PRIMARY KEY,
    nome VARCHAR(50) NOT NULL,
    valor_diaria DOUBLE NOT NULL
);

CREATE TABLE carro (
    placa VARCHAR(10) PRIMARY KEY,
    marca VARCHAR(50),
    modelo VARCHAR(50),
    cor VARCHAR(30),
    ano INT,
    tp_combustivel VARCHAR(20),
    qt_combustivel DOUBLE,
    quilometragem INT,
    cambio VARCHAR(20),
    disponibilidade BOOLEAN,

    categoria_id BIGINT,

    CONSTRAINT fk_carro_categoria
    FOREIGN KEY (categoria_id) REFERENCES categoria(id)
); 

CREATE TABLE locatario (
    cpf VARCHAR(14) PRIMARY KEY,
    nome VARCHAR(100),
    habilitacao VARCHAR(30),
    nascimento DATE,
    logradouro VARCHAR(100),
    numero INT,
    cep INT,
    cidade VARCHAR(50)
);

CREATE TABLE aluguel (
    id BIGINT AUTO_INCREMENT PRIMARY KEY,
    qt_dias INT,
    devolucao DATE,
    data_retirada DATE,
    valor_aluguel DOUBLE,
    valor_total DOUBLE,
    valor_combustivel DOUBLE,
    danos BOOLEAN,

    cpf_locatario VARCHAR(14),
    placa_carro VARCHAR(10),

    CONSTRAINT fk_aluguel_locatario
    FOREIGN KEY (cpf_locatario) REFERENCES locatario(cpf),

    CONSTRAINT fk_aluguel_carro
    FOREIGN KEY (placa_carro) REFERENCES carro(placa)
);

CREATE TABLE reparo (
    id BIGINT AUTO_INCREMENT PRIMARY KEY,

    placa_carro VARCHAR(10),
    entrada DATE NOT NULL,
    dias_reparo INT NOT NULL,
    defeito VARCHAR(200),
    valor_reparo DOUBLE,

    saida DATE NULL,

    CONSTRAINT fk_reparo_carro
    FOREIGN KEY (placa_carro) REFERENCES carro(placa)
);

    c.marca,
    c.modelo,
    l.nome,
    l.cpf,

    a.data_retirada

CREATE PROCEDURE listar_carros_disponiveis()
BEGIN
    DECLARE done INT DEFAULT 0;

    DECLARE v_placa VARCHAR(10);
    DECLARE v_marca VARCHAR(50);
    DECLARE v_modelo VARCHAR(50);
    DECLARE v_cor VARCHAR(30);
    DECLARE v_ano INT;
    DECLARE v_combustivel VARCHAR(20);
    DECLARE v_disponivel BOOLEAN;

    -- Cursor para percorrer carros
    DECLARE cur CURSOR FOR
        SELECT placa, marca, modelo, cor, ano, tp_combustivel, disponibilidade
        FROM carro;

    DECLARE CONTINUE HANDLER FOR NOT FOUND SET done = 1;

    -- Tabela temporária para retorno
    DROP TEMPORARY TABLE IF EXISTS tmp_carros;
    CREATE TEMPORARY TABLE tmp_carros (
        placa VARCHAR(10),
        marca VARCHAR(50),
        modelo VARCHAR(50),
        cor VARCHAR(30),
        ano INT,
        combustivel VARCHAR(20)
    );

    OPEN cur;

    read_loop: LOOP
        FETCH cur INTO v_placa, v_marca, v_modelo, v_cor, v_ano, v_combustivel, v_disponivel;

        IF done THEN
            LEAVE read_loop;
        END IF;

        -- Filtra apenas disponíveis
        IF v_disponivel = TRUE THEN
            INSERT INTO tmp_carros VALUES
            (v_placa, v_marca, v_modelo, v_cor, v_ano, v_combustivel);
        END IF;

    END LOOP;

    CLOSE cur;

    -- Retorno final
    SELECT * FROM tmp_carros;

END //

DELIMITER ;

CALL listar_carros_disponiveis();


END
